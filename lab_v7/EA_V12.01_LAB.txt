//| 🛡️ PARIDAD FORMULAS CORE SSOT (T-Score per-trade en Silicio)     |
//+------------------------------------------------------------------+
double CalcularTScoreTrade(double mfe_trade, double mae_trade, double prof_neta_trade, double t_gain, double t_loss, double spread_pips) {
   double fee = mfe_trade / (mfe_trade + MathAbs(mae_trade) + spread_pips + 1e-9);
   double fet = t_gain / MathMax(t_gain + t_loss, 1.0);
   double mae_piso = MathMax(MathAbs(mae_trade), spread_pips);
   double ivr_crudo = (prof_neta_trade - mae_trade) / (MathMax(t_loss, 1.0) * mae_piso + 1e-9);
   return (fee * fet * ((MathTanh(ivr_crudo) + 1.0) / 2.0));
}

bool EsNumeroValido(double v) {
   if(v != v) return false;                          
   if(v > 1.7e308 || v < -1.7e308) return false;      
   return true;
}

bool RangoValido01(double v) {
   return (EsNumeroValido(v) && v >= 0.0 && v <= 1.0);
}

bool PesosValidos(double &w[], int n) {
   bool hayNoCero = false;
   for(int i = 0; i < n; i++) {
      if(!EsNumeroValido(w[i])) return false;
      if(w[i] != 0.0) hayNoCero = true;
   }
   return hayNoCero; 
}

bool LabTamanoValido(CerebroML &ml) {
   if(ml.tam_modo != 1 && ml.tam_modo != 2) return false;
   if(!(ml.tam_act_med > 0.0) || !(ml.tam_a_is_med > 0.0)) return false;
   if(!(ml.tam_esc_min > 0.0) || !(ml.tam_esc_max >= ml.tam_esc_min)) return false;
   if(ml.tam_modo == 1) {
      if(!(ml.tam_med_pred > 0.0)) return false;
      for(int k = 0; k < 11; k++) { if(!(ml.tam_sd[k] > 0.0) || !MathIsValidNumber(ml.tam_w[k]) || !MathIsValidNumber(ml.tam_mu[k])) return false; }
   }
   return true;
}

bool EscribirTelemetriaCierre(MensajeAccion &msg, ulong position_id, int idxPar) {

   double precio_cierre = 0.0; datetime tiempo_cierre = 0;

   for(int i = 0; i < total_deals; i++) {
       ulong iter_deal = HistoryDealGetTicket(i);
       profit_broker_total  += HistoryDealGetDouble(iter_deal, DEAL_PROFIT);
       comision_total       += HistoryDealGetDouble(iter_deal, DEAL_COMMISSION);
       swap_total           += HistoryDealGetDouble(iter_deal, DEAL_SWAP);
       
       long deal_entry = HistoryDealGetInteger(iter_deal, DEAL_ENTRY);

           precio_cierre  = HistoryDealGetDouble(iter_deal, DEAL_PRICE);

       }
   }
   if(tiempo_cierre == 0) tiempo_cierre = TimeCurrent();

   string nombre_archivo = "Nexus_Telemetria_Trades.csv";

   if(handle == INVALID_HANDLE) return false;
   FileSeek(handle, 0, SEEK_END);
   
   if(!existe || FileSize(handle) == 0) {
      FileWrite(handle, "timestamp_cierre","simbolo","direccion","ticket","z_predicho","spread_pips","MAE_pips_interno","MFE_pips_interno","Profit_pips_interno","Profit_pips_broker","comision","swap","velas_vida","origen_cierre","fid","d_","theta","c_","t_pred_live");
   }
   
   FileWrite(handle, (long)tiempo_cierre, g_pares[idxPar].simbolo, (esToro ? "LONG" : "SHORT"), (long)position_id, z_pred_interno, g_pares[idxPar].spread_asumido_ia, mae_interno, mfe_interno, profit_interno, profit_broker_total, comision_total, swap_total, velas_vida, EnumToString(msg.origen), msg.diag_cierre.fid, msg.diag_cierre.d_, msg.diag_cierre.theta, msg.diag_cierre.c_, msg.diag_cierre.t_pred);

//| 🧮 MÓDULO 2: ÁLGEBRA O(1) Y ROBUST Z-SCORE C-ALLOC SAFE          |
//+------------------------------------------------------------------+
void CalcRegresion(double &y[], int n, double &a, double &v) {
   a = 0; v = 0; if(n < 3) return;
   double N_val = (double)(n - 1);
   double sx  = (N_val * (N_val + 1.0)) / 2.0; double sx2 = (N_val * (N_val + 1.0) * (2.0 * N_val + 1.0)) / 6.0; double sx3 = sx * sx;
   double sx4 = (N_val * (N_val + 1.0) * (2.0 * N_val + 1.0) * (3.0 * N_val * N_val + 3.0 * N_val - 1.0)) / 30.0;
   double D = sx4 * (sx2 * n - sx * sx) - sx3 * (sx3 * n - sx * sx2) + sx2 * (sx3 * sx - sx2 * sx2); if(D == 0.0) return;
   double sy=0, sxy=0, sx2y=0;
   for(int i=0; i<n; i++) {
      double x_actual = (double)((n - 1) - i); double yi = y[i]; sy += yi; sxy += yi * x_actual; sx2y += yi * (x_actual * x_actual);
   }
   double Da = sx2y * (sx2 * n - sx * sx) - sx3 * (sxy * n - sy * sx) + sx2 * (sxy * sx - sy * sx2);
   double Db = sx4 * (sxy * n - sy * sx) - sx2y * (sx3 * n - sx * sx2) + sx2 * (sx3 * sy - sxy * sx2);
   a = Da / D; double b = Db / D; v = 2.0 * a * (n - 1) + b;
}

double CalcRobustZScoreGeneric(double &hist[], int head, int pz_raw) {
   if(pz_raw < 2) return 0.0;
   int pz = (int)MathMin(pz_raw, 500); 
   
   double temp[500]; double sorted[500]; double devs[500];
   for(int i=0; i<pz; i++) { temp[i] = hist[(head + i) % 500]; sorted[i] = temp[i]; }
   
   for(int i=1; i<pz; i++) {
       double key = sorted[i]; int j = i - 1;
       while(j >= 0 && sorted[j] > key) { sorted[j + 1] = sorted[j]; j--; }
       sorted[j + 1] = key;
   }
   
   double mediana = (pz % 2 != 0) ? sorted[pz/2] : (sorted[(pz/2)-1] + sorted[pz/2]) / 2.0;
   
   for(int i=0; i<pz; i++) devs[i] = MathAbs(temp[i] - mediana);
   
   for(int i=1; i<pz; i++) {
       double key = devs[i]; int j = i - 1;
       while(j >= 0 && devs[j] > key) { devs[j + 1] = devs[j]; j--; }
       devs[j + 1] = key;
   }
   
   double mad = (pz % 2 != 0) ? devs[pz/2] : (devs[(pz/2)-1] + devs[pz/2]) / 2.0; if(mad < 1e-9) return 0.0; 
   return (hist[head] - mediana) / mad;
}

double Evaluar_Jefe_Riesgos_Ridge(double &X[], double &W[], double bias) { double Z = 0.0; for(int i=0; i<10; i++) { Z += W[i] * X[i]; } return Z + bias; }

// 📐 LAB-V7: actividad = Σ|ΔOpen| de 60 velas hasta la vela cerrada, en pips. Suma ascendente en el tiempo (desplazamiento 59 → 0).
double LabActividad(const string s, int shift_base, double pips_mult) {
   double act = 0.0;
   for(int m = 59; m >= 0; m--) {
      double o1 = iOpen(s, TF_Accion, shift_base + m);
      double o0 = iOpen(s, TF_Accion, shift_base + m + 1);
      if(o1 <= 0.0 || o0 <= 0.0) return -1.0;
      act += MathAbs(o1 - o0) * pips_mult;
   }
   return act;
}

// 📐 LAB-V7: escala de v_req. Mismo orden de operaciones que _lab_esc_A_nb / _lab_esc_B_nb de Fase_4_lab.py.
double LabEscalaTamano(CerebroML &ml, double &X[], double act) {
   if(ml.tam_modo == 0) return 1.0;
   double a = (act >= 0.0) ? (act / ml.tam_act_med) : ml.tam_a_is_med;
   double e;
   if(ml.tam_modo == 2) e = a / ml.tam_a_is_med;
   else {
      double t = 0.0;
      for(int k = 0; k < 11; k++) {
         double xk = (k < 10) ? X[k] : a;
         t += ml.tam_w[k] * ((xk - ml.tam_mu[k]) / ml.tam_sd[k]);
      }
      e = (t + ml.tam_intercept) / ml.tam_med_pred;
   }
   if(e < ml.tam_esc_min) e = ml.tam_esc_min;
   if(e > ml.tam_esc_max) e = ml.tam_esc_max;
   return e;
}

double ObtenerSatCausalOptimizada(double &extremos[], int cant, int dir) {
   double fallback_val = 10.0; if(dir == -1) fallback_val = -10.0; 
   if(cant < 5) return fallback_val;
   
   double sorted[100]; int limite = MathMin(cant, 100);
   for(int i = 0; i < limite; i++) sorted[i] = extremos[i];
   
   for(int i = 1; i < limite; i++) {
       double key = sorted[i]; int j = i - 1;
       while(j >= 0 && sorted[j] > key) { sorted[j + 1] = sorted[j]; j--; }
       sorted[j + 1] = key;
   }
   
   if(limite % 2 != 0) return sorted[limite/2]; 
   return (sorted[(limite/2)-1] + sorted[limite/2]) / 2.0; 
}

//+------------------------------------------------------------------+
//| 🧠 ACTUALIZADOR UNIVERSAL 4D (Edge Triggering)                   |
//+------------------------------------------------------------------+
bool ActualizarCinematicaCerebro(Cerebro &c, HistFPP &fpp, ENUM_MARCO_FSM marco, int res, int pz, ENUM_METRICA_BASE base, ENUM_METRICA_ACUMULADOR tension, double armado, double chasquido, ENUM_TIPO_BUSQUEDA busqueda, bool es_motor_salida, double &out_zscore_base, double &out_v_ml, double &out_a_ml, string debug_tag = "") {
   double crudo[500]; 
   int h_ptr = (marco == MARCO_FAST) ? fpp.head_fast : (marco == MARCO_MID ? fpp.head_mid : fpp.head_slow);
   if(marco == MARCO_FAST) { for(int i=0; i<res; i++) crudo[i] = fpp.fast[(h_ptr + i) % 500]; }
   else if(marco == MARCO_MID) { for(int i=0; i<res; i++) crudo[i] = fpp.mid[(h_ptr + i) % 500]; }
   else { for(int i=0; i<res; i++) crudo[i] = fpp.slow[(h_ptr + i) % 500]; }

   double val_base = 0; double val_tension_cruda = 0; double tension_final = 0; double val_v_cruda = 0; double val_a_cruda = 0; out_zscore_base = 0.0; out_v_ml = 0.0; out_a_ml = 0.0;

   if(g_velasContadas >= res) {
      double int1[500]; double int2[500]; 
      int1[res-1] = crudo[res-1]; for(int j=res-2; j>=0; j--) int1[j] = int1[j+1] + crudo[j];
      int2[res-1] = int1[res-1]; for(int j=res-2; j>=0; j--) int2[j] = int2[j+1] + int1[j];
      double a_fpp, v_fpp, a_i1, v_i1, a_i2, v_i2; CalcRegresion(crudo, res, a_fpp, v_fpp); CalcRegresion(int1, res, a_i1, v_i1); CalcRegresion(int2, res, a_i2, v_i2);

      switch(base) {
         case BASE_FPP_CRUDO: case BASE_FPP_V: case BASE_FPP_A: val_base = (base == BASE_FPP_CRUDO) ? crudo[0] : (base == BASE_FPP_V ? v_fpp : a_fpp); val_v_cruda = v_fpp; val_a_cruda = a_fpp; break;
         case BASE_INT_AREA: case BASE_INT_V: case BASE_INT_A: val_base = (base == BASE_INT_AREA) ? int1[0] : (base == BASE_INT_V ? v_i1 : a_i1); val_v_cruda = v_i1; val_a_cruda = a_i1; break;
         case BASE_INT2_AREA: case BASE_INT2_V: case BASE_INT2_A: val_base = (base == BASE_INT2_AREA) ? int2[0] : (base == BASE_INT2_V ? v_i2 : a_i2); val_v_cruda = v_i2; val_a_cruda = a_i2; break;
      }
      
      int t_enum = (int)tension; if(t_enum > 8) t_enum -= 9; 
      switch(t_enum) {
         case 0: val_tension_cruda=crudo[0]; break; case 1: val_tension_cruda=v_fpp; break; case 2: val_tension_cruda=a_fpp; break;
         case 3: val_tension_cruda=int1[0]; break; case 4: val_tension_cruda=v_i1; break; case 5: val_tension_cruda=a_i1; break;
         case 6: val_tension_cruda=int2[0]; break; case 7: val_tension_cruda=v_i2; break; case 8: val_tension_cruda=a_i2; break;
      }
   }

   c.head_base = (c.head_base - 1 + 500) % 500; c.hist_base[c.head_base] = val_base;
   c.head_tension = (c.head_tension - 1 + 500) % 500; c.hist_tension[c.head_tension] = val_tension_cruda;
   c.head_v = (c.head_v - 1 + 500) % 500; c.hist_v[c.head_v] = val_v_cruda;
   c.head_a = (c.head_a - 1 + 500) % 500; c.hist_a[c.head_a] = val_a_cruda;
   tension_final = val_tension_cruda; out_zscore_base = val_base; out_v_ml = val_v_cruda; out_a_ml = val_a_cruda;
   
   if(pz >= 2 && g_velasContadas >= pz) {
      if((int)tension > 8) { 
          tension_final = CalcRobustZScoreGeneric(c.hist_tension, c.head_tension, pz); 
          out_zscore_base = CalcRobustZScoreGeneric(c.hist_base, c.head_base, pz); 
          out_v_ml = CalcRobustZScoreGeneric(c.hist_v, c.head_v, pz);
          out_a_ml = CalcRobustZScoreGeneric(c.hist_a, c.head_a, pz);
      }
   }

   int nuevo_signo = (val_base > 0) ? 1 : ((val_base < 0) ? -1 : c.ultimo_signo_ciclo);
   if(nuevo_signo != c.ultimo_signo_ciclo && c.ultimo_signo_ciclo != 0) {
      if(c.ultimo_signo_ciclo == 1 && c.extremo_tension >= 0.0) {
         if(c.num_picos_max >= 100) { ArrayCopy(c.picos_max, c.picos_max, 0, 1, 99); c.picos_max[99] = c.extremo_tension; } else { c.picos_max[c.num_picos_max] = c.extremo_tension; c.num_picos_max++; }
      } else if(c.ultimo_signo_ciclo == -1 && c.extremo_tension <= 0.0) {

      }
      c.acum_ciclo = tension_final; c.extremo_tension = c.acum_ciclo;
   } else {
      c.acum_ciclo += tension_final;
      if(nuevo_signo > 0) c.extremo_tension = MathMax(c.extremo_tension, c.acum_ciclo);
      else if(nuevo_signo < 0) c.extremo_tension = MathMin(c.extremo_tension, c.acum_ciclo);
   }
   c.ultimo_signo_ciclo = nuevo_signo;

   bool condicion_logica = false;
   if(g_velasContadas > 500) {
      if(busqueda == BUSQUEDA_VALLE_ALCISTA && c.ultimo_signo_ciclo < 0) { 
         double sat_inf = ObtenerSatCausalOptimizada(c.picos_min, c.num_picos_min, -1);
         if(c.extremo_tension <= (sat_inf * armado) && c.acum_ciclo >= (c.extremo_tension * (1.0 - chasquido))) condicion_logica = true;
      } 
      else if(busqueda == BUSQUEDA_PICO_BAJISTA && c.ultimo_signo_ciclo > 0) { 
         double sat_sup = ObtenerSatCausalOptimizada(c.picos_max, c.num_picos_max, 1);
         if(c.extremo_tension >= (sat_sup * armado) && c.acum_ciclo <= (c.extremo_tension * (1.0 - chasquido))) condicion_logica = true;
      }
   }
   if(condicion_logica) c.edad_gatillo++; else c.edad_gatillo = 0;
   if(condicion_logica && (es_motor_salida || c.edad_gatillo <= UMBRAL_VELAS_EDAD)) return true;
   return false;
}

    datetime time_gmt = TimeGMT(); MqlDateTime gmt; TimeToStruct(time_gmt, gmt); bool is_dst = false;

    else if(gmt.mon == 11) { int previous_sunday = gmt.day - gmt.day_of_week; if(previous_sunday <= 0) is_dst = true; }

    if(ny.hour == 17 || (ny.hour == 16 && ny.min >= 55) || (ny.hour == 18 && ny.min <= 5)) return true;

                        max_hist = MathMax(max_hist, iHigh(g_pares[p].simbolo, TF_Accion, v)); 
                        min_hist = MathMin(min_hist, iLow(g_pares[p].simbolo, TF_Accion, v)); 
                        double op_v = iOpen(g_pares[p].simbolo, TF_Accion, v);
                        double prof_v = ((op_v - price_open) / g_pares[p].divisorPips) - g_pares[p].spread_asumido_ia;
                        if(prof_v > 0.0) tg++; else tl++;
                     } 
                  }
                  g_pares[p].max_toro = max_hist; g_pares[p].min_toro = min_hist;
                  g_pares[p].t_gain_toro = tg;   g_pares[p].t_loss_toro = tl;
                  
                  string gv_v = ObtenerGV_VREQ(t); string gv_z = ObtenerGV_ZPRED(t); string gv_t0 = ObtenerGV_THAT0(t);
                  g_pares[p].v_req_toro  = (GlobalVariableCheck(gv_v))  ? GlobalVariableGet(gv_v)  : 0.0;
                  g_pares[p].z_pred_toro = (GlobalVariableCheck(gv_z))  ? GlobalVariableGet(gv_z)  : 0.0;
                  g_pares[p].t_hat0_toro = (GlobalVariableCheck(gv_t0)) ? GlobalVariableGet(gv_t0) : 0.0;
               }
            }
            if(magic == g_pares[p].magicOso) {
               mt5_oso_vivo[p] = true; 
               if(g_pares[p].ticket_oso == 0) { 
                  g_pares[p].ticket_oso = t; g_pares[p].p_in_oso = price_open; g_pares[p].time_in_oso = time_open;
                  double max_hist = price_open; double min_hist = price_open; int shift_apertura = iBarShift(g_pares[p].simbolo, TF_Accion, time_open);
                  int tg = 0; int tl = 0;
                  if(shift_apertura > 0) { 
                     for(int v=1; v<=shift_apertura; v++) { 
                        max_hist = MathMax(max_hist, iHigh(g_pares[p].simbolo, TF_Accion, v)); 
                        min_hist = MathMin(min_hist, iLow(g_pares[p].simbolo, TF_Accion, v)); 
                        double op_v = iOpen(g_pares[p].simbolo, TF_Accion, v);
                        double prof_v = ((price_open - op_v) / g_pares[p].divisorPips) - g_pares[p].spread_asumido_ia;
                        if(prof_v > 0.0) tg++; else tl++;
                     } 
                  }
                  g_pares[p].max_oso = max_hist; g_pares[p].min_oso = min_hist;
                  g_pares[p].t_gain_oso = tg;    g_pares[p].t_loss_oso = tl;
                  
                  string gv_v_o = ObtenerGV_VREQ(t); string gv_z_o = ObtenerGV_ZPRED(t); string gv_t0_o = ObtenerGV_THAT0(t);
                  g_pares[p].v_req_oso  = (GlobalVariableCheck(gv_v_o))  ? GlobalVariableGet(gv_v_o)  : 0.0;
                  g_pares[p].z_pred_oso = (GlobalVariableCheck(gv_z_o))  ? GlobalVariableGet(gv_z_o)  : 0.0;
                  g_pares[p].t_hat0_oso = (GlobalVariableCheck(gv_t0_o)) ? GlobalVariableGet(gv_t0_o) : 0.0;
               }
            }
         }
      }
   }
   for(int p=0; p<g_numPares; p++) {
      if(g_pares[p].ticket_toro > 0 && !mt5_toro_vivo[p]) { 
         GlobalVariableDel(ObtenerGV_VREQ(g_pares[p].ticket_toro)); 
         GlobalVariableDel(ObtenerGV_ZPRED(g_pares[p].ticket_toro)); 
         GlobalVariableDel(ObtenerGV_THAT0(g_pares[p].ticket_toro));
         g_pares[p].ticket_toro = 0; g_pares[p].p_in_toro = 0; g_pares[p].max_toro = 0; g_pares[p].min_toro = 0; 
         g_pares[p].v_req_toro = 0;  g_pares[p].z_pred_toro = 0; g_pares[p].t_hat0_toro = 0;
         g_pares[p].t_gain_toro = 0; g_pares[p].t_loss_toro = 0;
      }
      if(g_pares[p].ticket_oso > 0 && !mt5_oso_vivo[p]) { 
         GlobalVariableDel(ObtenerGV_VREQ(g_pares[p].ticket_oso)); 
         GlobalVariableDel(ObtenerGV_ZPRED(g_pares[p].ticket_oso)); 
         GlobalVariableDel(ObtenerGV_THAT0(g_pares[p].ticket_oso));
         g_pares[p].ticket_oso = 0;  g_pares[p].p_in_oso = 0;  g_pares[p].max_oso = 0;  g_pares[p].min_oso = 0;  
         g_pares[p].v_req_oso = 0;   g_pares[p].z_pred_oso = 0;  g_pares[p].t_hat0_oso = 0;
         g_pares[p].t_gain_oso = 0;  g_pares[p].t_loss_oso = 0;
      }
   }
}

void EncolarCierresPorMagic(int idxPar, int magic, ENUM_ORIGEN_MENSAJE origen) {
   if((magic == g_pares[idxPar].magicToro && g_pares[idxPar].ticket_toro == 0) || (magic == g_pares[idxPar].magicOso && g_pares[idxPar].ticket_oso == 0)) return;

   DiagSnapshot diag_actual;
   bool esToroLocal = (magic == g_pares[idxPar].magicToro);
   if(esToroLocal) {
       diag_actual = g_pares[idxPar].diag_toro;
       diag_actual.p_in_freeze = g_pares[idxPar].p_in_toro; diag_actual.max_freeze = g_pares[idxPar].max_toro; diag_actual.min_freeze = g_pares[idxPar].min_toro;
       diag_actual.z_pred_freeze = g_pares[idxPar].z_pred_toro; diag_actual.time_in_freeze = g_pares[idxPar].time_in_toro;
   } else {
       diag_actual = g_pares[idxPar].diag_oso;
       diag_actual.p_in_freeze = g_pares[idxPar].p_in_oso; diag_actual.max_freeze = g_pares[idxPar].max_oso; diag_actual.min_freeze = g_pares[idxPar].min_oso;
       diag_actual.z_pred_freeze = g_pares[idxPar].z_pred_oso; diag_actual.time_in_freeze = g_pares[idxPar].time_in_oso;
   }

   for(int i=PositionsTotal()-1; i>=0; i--) {
      ulong t = PositionGetTicket(i);
      if(PositionSelectByTicket(t) && PositionGetString(POSITION_SYMBOL)==g_pares[idxPar].simbolo && (int)PositionGetInteger(POSITION_MAGIC)==magic) {
         bool yaEncolado = false; for(int j=0; j<g_numMensajes; j++) { if(g_mensajes[j].ticketCierre == t && !g_mensajes[j].ejecutado) { yaEncolado = true; break; } }
         if(!yaEncolado) EncolarMensajeCierreConDiag(idxPar, origen, t, magic, diag_actual);
      }
   }
}

void EjecutarBuzon() {
   if(g_numMensajes == 0) return;
   if(!TerminalInfoInteger(TERMINAL_CONNECTED) || !TerminalInfoInteger(TERMINAL_TRADE_ALLOWED) || !MQLInfoInteger(MQL_TRADE_ALLOWED)) return; 

   double margen_requerido_total = 0.0; double maxLoteEnCola = 0.0;
   for(int i=0; i<g_numMensajes; i++) {
      if(!g_mensajes[i].ejecutado && !g_mensajes[i].loteEscalado && (g_mensajes[i].accion == MSG_APERTURA_COMPRA || g_mensajes[i].accion == MSG_APERTURA_VENTA)) {
         if(g_mensajes[i].lote > maxLoteEnCola) maxLoteEnCola = g_mensajes[i].lote;
         double mrg = 0.0; ENUM_ORDER_TYPE tipo = (g_mensajes[i].accion == MSG_APERTURA_COMPRA) ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
         double precio = (tipo == ORDER_TYPE_BUY) ? SymbolInfoDouble(g_mensajes[i].simbolo, SYMBOL_ASK) : SymbolInfoDouble(g_mensajes[i].simbolo, SYMBOL_BID);
         if(!OrderCalcMargin(tipo, g_mensajes[i].simbolo, g_mensajes[i].lote, precio, mrg)) mrg = AccountInfoDouble(ACCOUNT_MARGIN_FREE) * 2.0; 
         margen_requerido_total += mrg;
      }
   }

   double factorMaxLote = (maxLoteEnCola > LoteMaximo) ? (LoteMaximo / maxLoteEnCola) : 1.0; double margen_libre = AccountInfoDouble(ACCOUNT_MARGIN_FREE); double margen_permitido = margen_libre * 0.85; double factorMargen = 1.0;
   if(margen_requerido_total > margen_permitido && margen_requerido_total > 0) factorMargen = margen_permitido / margen_requerido_total;
   double factorEscalarFinal = MathMin(factorMaxLote, factorMargen);

   for(int i=0; i<g_numMensajes; i++) {
      if(!g_mensajes[i].loteEscalado && (g_mensajes[i].accion == MSG_APERTURA_COMPRA || g_mensajes[i].accion == MSG_APERTURA_VENTA)) {
         double loteBalanceado = g_mensajes[i].lote * factorEscalarFinal;
         double step = SymbolInfoDouble(g_mensajes[i].simbolo, SYMBOL_VOLUME_STEP); if(step <= 0.0) step = 0.01; double minL = SymbolInfoDouble(g_mensajes[i].simbolo, SYMBOL_VOLUME_MIN); if(minL <= 0.0) minL = 0.01;
         g_mensajes[i].lote = NormalizeDouble(MathRound(MathMax(minL, loteBalanceado)/step)*step, 2); g_mensajes[i].loteEscalado = true;
      }
   }

   bool esRollover = (Bloquear_Rollover && EsRolloverToxicoNY());
   for(int i=0; i<g_numMensajes; i++) {
      if(g_mensajes[i].ejecutado) {
          if(g_mensajes[i].accion == MSG_CIERRE_SAR && !g_mensajes[i].telemetria_extraida) {
              g_mensajes[i].reintentos++;
              g_mensajes[i].telemetria_extraida = EscribirTelemetriaCierre(g_mensajes[i], g_mensajes[i].ticketCierre, g_mensajes[i].idxPar);
              if(!g_mensajes[i].telemetria_extraida && g_mensajes[i].reintentos >= 50) g_mensajes[i].telemetria_extraida = true; 
          }
          continue;
      }
      
      int idx = g_mensajes[i].idxPar; bool esNuevaOrden = (g_mensajes[i].accion != MSG_CIERRE_SAR);
      
      if(esNuevaOrden) {
         int max_segundos_ttl = UMBRAL_VELAS_EDAD * PeriodSeconds(TF_Accion);
         if(TimeCurrent() - g_mensajes[i].timestamp_creacion > max_segundos_ttl) { g_mensajes[i].ejecutado = true; continue; }
         if(esRollover) continue; 
         double spreadActual = (double)SymbolInfoInteger(g_mensajes[i].simbolo, SYMBOL_SPREAD) * SymbolInfoDouble(g_mensajes[i].simbolo, SYMBOL_POINT);
         double spreadPips = spreadActual / g_pares[idx].divisorPips;
         if(spreadPips > Max_Spread_Pips) continue; 
      }
      
      g_mensajes[i].reintentos++;
      
      if(!esNuevaOrden) { 
         ulong t = g_mensajes[i].ticketCierre; 
         if(PositionSelectByTicket(t)) {
            if(trade.PositionClose(t)) { 
                uint ret = trade.ResultRetcode(); ulong deal = trade.ResultDeal(); 
                if(ret == TRADE_RETCODE_DONE || ret == TRADE_RETCODE_PLACED || ret == TRADE_RETCODE_DONE_PARTIAL || deal > 0) {
                    g_mensajes[i].ejecutado = true;
                    g_mensajes[i].reintentos = 0; 
                    g_mensajes[i].telemetria_extraida = EscribirTelemetriaCierre(g_mensajes[i], t, idx);
                }
            }
         } else { 
             g_mensajes[i].ejecutado = true; 
             g_mensajes[i].reintentos = 0; 
             g_mensajes[i].telemetria_extraida = EscribirTelemetriaCierre(g_mensajes[i], t, idx);
         }
      }
      else {
         trade.SetExpertMagicNumber(g_mensajes[i].magicNumber);
         string comentario = "NX_" + DoubleToString(g_mensajes[i].v_req_trx, 4);
         bool success = (g_mensajes[i].accion == MSG_APERTURA_COMPRA) ? trade.Buy(g_mensajes[i].lote, g_mensajes[i].simbolo, 0, 0, 0, comentario) : trade.Sell(g_mensajes[i].lote, g_mensajes[i].simbolo, 0, 0, 0, comentario);
         
         if(success) { 
             uint ret = trade.ResultRetcode(); ulong deal = trade.ResultDeal(); 
             if(ret == TRADE_RETCODE_DONE || ret == TRADE_RETCODE_PLACED || ret == TRADE_RETCODE_DONE_PARTIAL || deal > 0) { 
                 g_mensajes[i].ejecutado = true; 
                 g_mensajes[i].telemetria_extraida = true; 
                 ulong trd_tck = trade.ResultOrder(); 
                 if(trd_tck <= 0) trd_tck = trade.ResultDeal(); 
                 if(trd_tck > 0) {
                     if(g_mensajes[i].v_req_trx != 0.0)  GlobalVariableSet(ObtenerGV_VREQ(trd_tck), g_mensajes[i].v_req_trx);
                     if(g_mensajes[i].z_pred_trx != 0.0) GlobalVariableSet(ObtenerGV_ZPRED(trd_tck), g_mensajes[i].z_pred_trx);
                     if(g_mensajes[i].t_hat0_trx != 0.0) GlobalVariableSet(ObtenerGV_THAT0(trd_tck), g_mensajes[i].t_hat0_trx);
                 }
             } 
         }
      }
      if(!g_mensajes[i].ejecutado && g_mensajes[i].reintentos >= 10) { g_mensajes[i].ejecutado = true; g_mensajes[i].telemetria_extraida = true; g_cached_positions = -1; }
   }
   
   MensajeAccion pendientes[]; int conteoPendientes = 0; ArrayResize(pendientes, g_numMensajes);
   for(int i=0; i<g_numMensajes; i++) { 
       bool purgar = g_mensajes[i].ejecutado;
       if(g_mensajes[i].accion == MSG_CIERRE_SAR && !g_mensajes[i].telemetria_extraida) purgar = false; 
       if(!purgar) { pendientes[conteoPendientes] = g_mensajes[i]; conteoPendientes++; } 
   }
   g_numMensajes = conteoPendientes; ArrayResize(g_mensajes, g_numMensajes);
   for(int i=0; i<g_numMensajes; i++) g_mensajes[i] = pendientes[i];
}

bool ExisteOrdenPendiente(int idxPar, ENUM_TIPO_MENSAJE accion) {
   for(int i=0; i<g_numMensajes; i++) { if(!g_mensajes[i].ejecutado && g_mensajes[i].idxPar == idxPar && g_mensajes[i].accion == accion) return true; } return false;
}

void AnalizarMercado(datetime target_time) {
   int paresActivos = 0; bool parValido[]; ArrayResize(parValido, g_numPares); ArrayInitialize(parValido, false);
   for(int i=0; i<g_numPares; i++) {
      string s = g_pares[i].simbolo; if(!g_pares[i].esOperable || !g_pares[i].cfg.configurado) continue;
      int shift_base = iBarShift(s, TF_Accion, target_time, false); if(shift_base < 0) continue; 
      
      bool es_tiempo_real_local = (target_time >= iTime(s, TF_Accion, 1));
      if(es_tiempo_real_local) { 
         bool opsToro = (g_pares[i].ticket_toro > 0); bool opsOso  = (g_pares[i].ticket_oso > 0);
         if(opsToro || opsOso) {
            double high_v = iHigh(s, TF_Accion, shift_base); double low_v  = iLow(s, TF_Accion, shift_base);
            if(opsToro && g_pares[i].max_toro > 0.0 && high_v > 0) { g_pares[i].max_toro = MathMax(g_pares[i].max_toro, high_v); g_pares[i].min_toro = MathMin(g_pares[i].min_toro, low_v); }
            if(opsOso && g_pares[i].max_oso > 0.0 && high_v > 0) { g_pares[i].max_oso = MathMax(g_pares[i].max_oso, high_v); g_pares[i].min_oso = MathMin(g_pares[i].min_oso, low_v); }
         }
      } 
      double bid_sincronizado = iClose(s, TF_Accion, shift_base); if (bid_sincronizado <= 0.0) continue;
      
      bool serieCompleta = true;
      for(int k=0; k<g_numTFs; k++) {
         int lookback_velas = MathMax(1, (PeriodSeconds(g_tfsArray[k]) / PeriodSeconds(TF_Accion)) * g_lookbackArray[k]);
         datetime time_pasado = target_time - (lookback_velas * PeriodSeconds(TF_Accion));
         int shift_pasado = iBarShift(s, TF_Accion, time_pasado, false); if(shift_pasado < 0) { serieCompleta = false; break; }
         double opN_sincronizado = iClose(s, TF_Accion, shift_pasado); if(opN_sincronizado <= 0.0) { serieCompleta = false; break; }
         g_deltaNominal[i].cols[k] = (bid_sincronizado - opN_sincronizado) / g_pares[i].divisorPips;
      }
      if(serieCompleta) { parValido[i] = true; paresActivos++; }
   }
   
   int umbralMinimo = MathMax(1, (int)(g_numPares * 0.70)); if(paresActivos < umbralMinimo) return; 
   double min_div_cuad = 1.0 / (g_numPares * g_numPares); double fppTemp[]; ArrayResize(fppTemp, g_numPares); 
   for(int k=0; k<g_numTFs; k++) {
      double pipsTotales = 0; for(int i=0; i<g_numPares; i++) { if(parValido[i]) pipsTotales += MathAbs(g_deltaNominal[i].cols[k]); }
      double sumaFtempNeta = 0; ArrayInitialize(fppTemp, 0.0); double pipsTotalesSafe = (pipsTotales > 0) ? (pipsTotales * g_numPares) : 1.0; 
      for(int i=0; i<g_numPares; i++) { if(parValido[i]) { fppTemp[i] = g_deltaNominal[i].cols[k] / pipsTotalesSafe; sumaFtempNeta += fppTemp[i]; } }
      double absSumaFtemp = MathAbs(sumaFtempNeta); double divCuad = MathMax(absSumaFtemp * absSumaFtemp, min_div_cuad);
      
      double fppOriginal[]; ArrayResize(fppOriginal, g_numPares); ArrayInitialize(fppOriginal, 0.0);
      double sumaAbsOriginal = 0.0; double sumaRelOriginal = 0.0;
      for(int i=0; i<g_numPares; i++) {
         if(!parValido[i]) continue;
         fppOriginal[i] = fppTemp[i] / divCuad; sumaAbsOriginal += MathAbs(fppOriginal[i]); sumaRelOriginal += fppOriginal[i];
      }
      double k_kuramoto = (sumaAbsOriginal > 0.0) ? (sumaRelOriginal / sumaAbsOriginal) : 0.0;
      double sigma_acople = 0.25; 
      for(int i=0; i<g_numPares; i++) {
         if(!parValido[i]) continue;
         double diferencia_fase = k_kuramoto - fppOriginal[i]; double fppVal = fppOriginal[i] + (sigma_acople * MathTanh(diferencia_fase));
         if(k==0) { g_pares[i].fpp.head_fast = (g_pares[i].fpp.head_fast - 1 + 500) % 500; g_pares[i].fpp.fast[g_pares[i].fpp.head_fast] = fppVal; }
         if(k==1) { g_pares[i].fpp.head_mid = (g_pares[i].fpp.head_mid - 1 + 500) % 500; g_pares[i].fpp.mid[g_pares[i].fpp.head_mid] = fppVal; }
         if(k==2) { g_pares[i].fpp.head_slow = (g_pares[i].fpp.head_slow - 1 + 500) % 500; g_pares[i].fpp.slow[g_pares[i].fpp.head_slow] = fppVal; }
      }
   }
   g_velasContadas++;
    
   datetime ultima_vela_cerrada = iTime(_Symbol, TF_Accion, 1); bool es_tiempo_real = (target_time >= ultima_vela_cerrada); 
   if(es_tiempo_real) RegistrarEquitySnapshot(); 
   double capital = AccountInfoDouble(ACCOUNT_BALANCE); double lote_base_por_equidad = (capital / 1000.0) * Riesgo_Base_Por_1000;

   for(int i=0; i<g_numPares; i++) {
      if(!parValido[i] || !g_pares[i].esOperable || !g_pares[i].cfg.configurado) continue; 
      
      string s = g_pares[i].simbolo;
      double zscore_base_toro_in=0, zscore_base_toro_out=0, zscore_base_oso_in=0, zscore_base_oso_out=0; double v_in_toro=0, a_in_toro=0, v_out_toro=0, a_out_toro=0; double v_in_oso=0, a_in_oso=0, v_out_oso=0, a_out_oso=0;     

      bool gatillo_toro_in = ActualizarCinematicaCerebro(g_pares[i].c_toro_in, g_pares[i].fpp, g_pares[i].cfg.toro.marco_in, (int)g_pares[i].cfg.toro.res_in, g_pares[i].cfg.toro.pz_in, g_pares[i].cfg.toro.base_in, g_pares[i].cfg.toro.tension_in, g_pares[i].cfg.toro.armado_in, g_pares[i].cfg.toro.chasquido_in, BUSQUEDA_VALLE_ALCISTA, false, zscore_base_toro_in, v_in_toro, a_in_toro, s + "-TORO-IN");

            X_ML[4] = zscore_base_oso_out; X_ML[5] = v_out_oso; X_ML[6] = a_out_oso; X_ML[7] = MathSin(MathArctan2(X_ML[5], X_ML[4])); 
            X_ML[8] = z_macro_1_oso; X_ML[9] = z_macro_2_oso;
            
            double Z_pred = Evaluar_Jefe_Riesgos_Ridge(X_ML, g_pares[i].ml_oso.w, g_pares[i].ml_oso.bias);
            if(Z_pred < g_pares[i].ml_oso.umbral) autorizado_ml = false; 
            else { 
                zpred_calc_oso = Z_pred; 
                vreq_calc_oso = ((Z_pred * g_pares[i].ml_oso.mad_v) + g_pares[i].ml_oso.mediana_v) * g_pares[i].ml_oso.ratio_captura; 
                if(g_pares[i].ml_oso.tam_modo != 0) vreq_calc_oso = vreq_calc_oso * LabEscalaTamano(g_pares[i].ml_oso, X_ML, LabActividad(s, 1, MathRound(1.0 / g_pares[i].divisorPips))); // LAB-V7
                
                if(g_pares[i].ml_oso.cabeza_t_activa) {
                    double T_raw_oso = Evaluar_Jefe_Riesgos_Ridge(X_ML, g_pares[i].ml_oso.w_T, g_pares[i].ml_oso.bias_T);
                    if(T_raw_oso < g_pares[i].ml_oso.t_min_admision) autorizado_ml = false;
                    else that0_calc_oso = MathMin(MathMax(T_raw_oso, 0.0), 1.0);
                }
            }
         }
         if(autorizado_ml) {
             g_pares[i].t_hat0_oso = that0_calc_oso;
             g_pares[i].t_gain_oso = 0; g_pares[i].t_loss_oso = 0;
             EncolarMensaje(i, MSG_APERTURA_VENTA, ORIGEN_ENTRADA_SAR, 0, lote_base_por_equidad * MathAbs(zscore_base_oso_in), g_pares[i].magicOso, zpred_calc_oso, vreq_calc_oso, that0_calc_oso);
         }
      } else if(opsOso && salir_oso) EncolarCierresPorMagic(i, g_pares[i].magicOso, origen_cierre_oso);
   }
}

ENUM_TIMEFRAMES StringToTF(string s) {
   StringTrimLeft(s); StringTrimRight(s); 
   if(s == "PERIOD_M1")  return PERIOD_M1;  if(s == "PERIOD_M5")  return PERIOD_M5;
   if(s == "PERIOD_M15") return PERIOD_M15; if(s == "PERIOD_M30") return PERIOD_M30;
   if(s == "PERIOD_H1")  return PERIOD_H1;  if(s == "PERIOD_H2")  return PERIOD_H2;
   if(s == "PERIOD_H4")  return PERIOD_H4;  if(s == "PERIOD_H6")  return PERIOD_H6;
   if(s == "PERIOD_D1")  return PERIOD_D1;  
   return PERIOD_D1; 
}

int OnInit() {
   Print(">>> NexusHybrid V12.01 DUAL CONTINUOUS ML CORE - ADAPTATIVE SSOT 3.3 CERTIFIED <<<");
   
   if(GlobalVariableCheck(g_mutex_name)) {
      double marca_previa = GlobalVariableGet(g_mutex_name);
      double antiguedad = (double)TimeLocal() - marca_previa;
      if(antiguedad < MUTEX_TTL_SEGUNDOS) {
         Alert("🚨 CRÍTICO: Mutex MQL — instancia activa detectada (antigüedad " + DoubleToString(antiguedad,0) + "s).");
         return INIT_FAILED;
      } else {
         Telemetria(LOG_LEVEL_0_PROD, "⚠️ [V12.01] Mutex obsoleto detectado (antigüedad " + DoubleToString(antiguedad,0) + "s) — liberando.");
         GlobalVariableDel(g_mutex_name);
      }
   }
   GlobalVariableTemp(g_mutex_name); GlobalVariableSet(g_mutex_name, (double)TimeLocal());
   if(AccountInfoInteger(ACCOUNT_MARGIN_MODE) != ACCOUNT_MARGIN_MODE_RETAIL_HEDGING) return INIT_FAILED; 
   
   StringSplit(Pares28, ',', g_paresArray); StringSplit(MarcosTemporales, ',', g_tfsStrings);
   g_numPares = ArraySize(g_paresArray); g_numTFs = ArraySize(g_tfsStrings); ArrayResize(g_tfsArray, g_numTFs); for(int k=0; k<g_numTFs; k++) g_tfsArray[k] = StringToTF(g_tfsStrings[k]);
   string lbStr[]; StringSplit(Lookback_Velas_FPP, ',', lbStr); ArrayResize(g_lookbackArray, g_numTFs); for(int k=0; k<g_numTFs; k++) g_lookbackArray[k] = (k < ArraySize(lbStr)) ? (int)StringToInteger(lbStr[k]) : 1; 
   ArrayResize(g_pares, g_numPares); ArrayResize(g_deltaNominal, g_numPares);
   
   for(int i=0; i<g_numPares; i++) {

      int hash_sym = ObtenerHashSimbolo(g_pares[i].simbolo);
      
      g_pares[i].magicToro = 3000000 + (hash_sym % 100000) * 10 + 1; 
      g_pares[i].magicOso  = 3000000 + (hash_sym % 100000) * 10 + 2;
      
      g_pares[i].ticket_toro = 0; g_pares[i].ticket_oso = 0; 
      g_pares[i].fpp.head_fast = 0; g_pares[i].fpp.head_mid = 0; g_pares[i].fpp.head_slow = 0;
      ArrayInitialize(g_pares[i].fpp.fast, 0.0); ArrayInitialize(g_pares[i].fpp.mid, 0.0); ArrayInitialize(g_pares[i].fpp.slow, 0.0);
      
      InitCerebro(g_pares[i].c_toro_in); InitCerebro(g_pares[i].c_toro_out);
      InitCerebro(g_pares[i].c_oso_in); InitCerebro(g_pares[i].c_oso_out);
      InitCerebro(g_pares[i].c_toro_macro_1); InitCerebro(g_pares[i].c_toro_macro_2);
      InitCerebro(g_pares[i].c_oso_macro_1); InitCerebro(g_pares[i].c_oso_macro_2);
      
      g_pares[i].ml_toro.activado = false; g_pares[i].ml_oso.activado = false;

      g_pares[i].ml_oso.cabeza_t_activa  = false; g_pares[i].ml_oso.t_min_admision = 0.0;
      g_pares[i].ml_toro.tam_modo = 0; g_pares[i].ml_oso.tam_modo = 0;   // LAB-V7: Ridge de tamaño apagado por defecto

void OnDeinit(const int reason) { EventKillTimer(); GlobalVariableDel(g_mutex_name); }

void OnTimer() {
   if(g_boot_state == BOOT_READY && TimeLocal() - g_ultimo_heartbeat >= 60) {
      g_ultimo_heartbeat = TimeLocal();
      GlobalVariableSet(g_mutex_name, (double)TimeLocal());
      for(int p=0; p<g_numPares; p++) {
         if(g_pares[p].ticket_toro > 0) { 
             if(GlobalVariableCheck(ObtenerGV_VREQ(g_pares[p].ticket_toro))) g_pares[p].v_req_toro = GlobalVariableGet(ObtenerGV_VREQ(g_pares[p].ticket_toro)); 
             if(GlobalVariableCheck(ObtenerGV_ZPRED(g_pares[p].ticket_toro))) g_pares[p].z_pred_toro = GlobalVariableGet(ObtenerGV_ZPRED(g_pares[p].ticket_toro)); 
             if(GlobalVariableCheck(ObtenerGV_THAT0(g_pares[p].ticket_toro))) g_pares[p].t_hat0_toro = GlobalVariableGet(ObtenerGV_THAT0(g_pares[p].ticket_toro));
         }
         if(g_pares[p].ticket_oso > 0) { 
             if(GlobalVariableCheck(ObtenerGV_VREQ(g_pares[p].ticket_oso))) g_pares[p].v_req_oso = GlobalVariableGet(ObtenerGV_VREQ(g_pares[p].ticket_oso)); 
             if(GlobalVariableCheck(ObtenerGV_ZPRED(g_pares[p].ticket_oso))) g_pares[p].z_pred_oso = GlobalVariableGet(ObtenerGV_ZPRED(g_pares[p].ticket_oso)); 
             if(GlobalVariableCheck(ObtenerGV_THAT0(g_pares[p].ticket_oso))) g_pares[p].t_hat0_oso = GlobalVariableGet(ObtenerGV_THAT0(g_pares[p].ticket_oso));
         }
      }
   }
   
   if(g_boot_state != BOOT_READY) {
      if(g_boot_state == BOOT_REQUEST_DATA) { for(int i=0; i<g_numPares; i++) { datetime dummy[1]; CopyTime(g_pares[i].simbolo, TF_Accion, 0, BUFFER_CACHE, dummy); } g_boot_state = BOOT_WAIT_SYNC; return; }
      if(g_boot_state == BOOT_WAIT_SYNC) {
         bool todoListo = true; int tiempo_espera = (int)(TimeLocal() - g_ultimo_heartbeat);
         for(int i=0; i<g_numPares; i++) {
            if(!g_pares[i].esOperable) continue;
            bool sinc = MQLInfoInteger(MQL_TESTER) ? (iBars(g_pares[i].simbolo, TF_Accion) >= BUFFER_CACHE) : (SeriesInfoInteger(g_pares[i].simbolo, TF_Accion, SERIES_SYNCHRONIZED));
            if(!sinc) { if(tiempo_espera > 30) g_pares[i].esOperable = false; else { todoListo = false; break; } }
         }
         if(!todoListo) return; g_boot_state = BOOT_PROCESS_CHUNKS; return;
      }
      if(g_boot_state == BOOT_PROCESS_CHUNKS) {
         int cuota_restante = CHUNK_SIZE;
         while(g_shift_actual >= 1 && cuota_restante > 0) { AnalizarMercado(iTime(_Symbol, TF_Accion, g_shift_actual)); g_shift_actual--; cuota_restante--; }
         if(g_shift_actual < 1) { g_ultimaVelaAccion = iTime(_Symbol, TF_Accion, 0); g_boot_state = BOOT_READY; Telemetria(LOG_LEVEL_0_PROD, "🚀 PRECARGA AL 100%. Motor Cuántico DUAL INSTITUCIONAL ONLINE."); }
         return; 
      }
   }
   g_timer_buzon++; if(g_timer_buzon >= 3) { g_timer_buzon = 0; if(g_numMensajes > 0) EjecutarBuzon(); }
}

void OnTick() {
   if(g_boot_state != BOOT_READY) { if(MQLInfoInteger(MQL_TESTER)) OnTimer(); return; }
   int current_positions = PositionsTotal(); if(current_positions != g_cached_positions) { SincronizarEstadosReales(); g_cached_positions = current_positions; }
   datetime velaActual = iTime(_Symbol, TF_Accion, 0);
   if(velaActual != g_ultimaVelaAccion) {
      int velas_perdidas = 1; if(g_ultimaVelaAccion > 0) { velas_perdidas = iBarShift(_Symbol, TF_Accion, g_ultimaVelaAccion); if(velas_perdidas <= 0) velas_perdidas = 1; }
      if(velas_perdidas > 500) velas_perdidas = 500; SincronizarEstadosReales(); 
      for(int shift = velas_perdidas; shift >= 1; shift--) { AnalizarMercado(iTime(_Symbol, TF_Accion, shift)); }
      g_ultimaVelaAccion = velaActual; EjecutarBuzon(); 
   }
}
